#' @examples
#'

#' node_tour <- get_node_tour(forest)
#'

#'
#' # repeat until the end has been reached

#'   # increase the counter
#'   counter <- counter + 1
#'
#'   # move to the next node
#'   node_tour$step()
#' }
#'
#' # show the node count
#' counter
#'
#' # show the forest (and the number of its nodes)
#' forest

#'   `vignette("node_tour")`
